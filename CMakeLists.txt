cmake_minimum_required(VERSION 3.20)
project(Emergence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(city_core sim/city.cpp sim/life.cpp sim/economy.cpp geo/terrain.cpp replay/checkpoint.cpp)
target_include_directories(city_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(city_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
set_target_properties(city_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
enable_testing()
add_executable(city_tests tests/city_tests.cpp)
target_link_libraries(city_tests PRIVATE city_core)
add_test(NAME city_slice COMMAND city_tests)
add_executable(city_profile tests/profile.cpp)
target_link_libraries(city_profile PRIVATE city_core)
option(CITY_EXTENSION "Build the macOS Godot adapter" ON)
if(CITY_EXTENSION)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/godot-cpp/CMakeLists.txt")
        message(FATAL_ERROR "Clone godot-cpp tag godot-4.4.1-stable into vendor/godot-cpp or set CITY_EXTENSION=OFF.")
    endif()
    set(GODOTCPP_TARGET "template_release" CACHE STRING "Godot C++ target")
    set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/extension/profile.json" CACHE PATH "")
    add_subdirectory(vendor/godot-cpp EXCLUDE_FROM_ALL)
    add_library(emergence SHARED extension/bridge.cpp)
    target_link_libraries(emergence PRIVATE city_core godot::cpp)
    set_target_properties(emergence PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/game/bin")
endif()
